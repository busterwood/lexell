# INFORMATION_SCHEMA lists no NOT NULL constraints (PostgreSQL 18 has them; SQL Server does not)
@once CREATE TABLE isnn (id int NOT NULL PRIMARY KEY, a int NOT NULL, b int CONSTRAINT ckisb CHECK (b > 0), c varchar(5) NOT NULL)
SELECT CONSTRAINT_TYPE, CASE WHEN CONSTRAINT_TYPE = 'PRIMARY KEY' THEN 'pk' ELSE CONSTRAINT_NAME END AS n FROM INFORMATION_SCHEMA.TABLE_CONSTRAINTS WHERE TABLE_NAME = 'isnn' ORDER BY CONSTRAINT_TYPE, n
SELECT CONSTRAINT_NAME FROM INFORMATION_SCHEMA.CHECK_CONSTRAINTS WHERE CONSTRAINT_NAME LIKE '%isnn%' OR CONSTRAINT_NAME = 'ckisb' ORDER BY CONSTRAINT_NAME
SELECT COLUMN_NAME FROM INFORMATION_SCHEMA.CONSTRAINT_COLUMN_USAGE WHERE TABLE_NAME = 'isnn' ORDER BY COLUMN_NAME
SELECT COLUMN_NAME, IS_NULLABLE FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_NAME = 'isnn' ORDER BY ORDINAL_POSITION
