# MERGE, compared with PostgreSQL (tests/lexdb/gen_pg.py merge).
create table tgt (id int primary key, name text, qty int not null default 0, note text)
insert into tgt values (1, 'a', 10, null), (2, 'b', 20, null), (3, 'c', 30, null)
create table src (id int, name text, qty int)
insert into src values (2, 'B', 5), (3, 'C', 7), (4, 'D', 9), (5, 'E', 11)
# an upsert: matched rows updated, the others inserted
merge into tgt t using src s on t.id = s.id when matched then update set name = s.name, qty = t.qty + s.qty when not matched then insert (id, name, qty) values (s.id, s.name, s.qty)
select * from tgt order by id
# no alias, INTO's table name qualifying; a column list left out; DEFAULT in values
merge into tgt using (values (6, 'F', 1)) as v(id, name, qty) on tgt.id = v.id when not matched then insert values (v.id, v.name, default)
select * from tgt where id = 6
# conditional clauses: the first one whose condition holds is the one used
merge into tgt t using src s on t.id = s.id when matched and s.qty > 10 then delete when matched and s.qty > 6 then update set note = 'big' when matched then update set note = 'small' when not matched and s.qty < 10 then insert (id, name) values (s.id, s.name)
select * from tgt order by id
# DO NOTHING, and a matched row with no clause for it
merge into tgt t using src s on t.id = s.id when matched and t.id = 2 then do nothing when matched and t.id = 3 then update set note = 'three'
select * from tgt where id in (2, 3) order by id
merge into tgt t using src s on t.id = s.id when not matched then do nothing
merge into tgt t using src s on t.id = s.id when matched then do nothing
# DELETE
merge into tgt t using (select 1 as id union all select 2) s on t.id = s.id when matched then delete
select * from tgt order by id
# a subquery source, VALUES, a source that is a table joined, an aggregate
merge into tgt t using (select id, name from src where id > 3) s on t.id = s.id when matched then update set name = lower(s.name) when not matched then insert (id, name) values (s.id + 100, s.name)
select * from tgt order by id
merge into tgt t using (values (3, 'x'), (77, 'y')) as v (id, name) on t.id = v.id when matched then update set name = v.name when not matched then insert (id, name) values (v.id, v.name)
select * from tgt order by id
merge into tgt t using (select name, sum(qty) as q from src group by name) s on t.name = lower(s.name) when matched then update set qty = s.q
select * from tgt order by id
# the source rows many for one target row: an error when it would be changed twice

merge into tgt t using dup d on t.id = d.id when matched and d.name = 'p' then update set name = 'only p'

merge into tgt t using dup d on t.id = d.id when matched then do nothing when not matched then insert (id, name) values (d.id, d.name)
select * from tgt order by id
# two source rows inserting the same new key: a unique violation, nothing done
merge into tgt t using (values (50, 'a'), (50, 'b')) as v(id, name) on t.id = v.id when not matched then insert (id, name) values (v.id, v.name)
select count(*) from tgt where id = 50
# an empty source, and a source with no match
merge into tgt t using (select * from src where false) s on t.id = s.id when matched then delete when not matched then insert (id) values (s.id)
merge into tgt t using (values (900, 'z')) as v(id, name) on t.id = v.id when matched then delete
merge into tgt t using (values (900, 'z')) as v(id, name) on t.id = v.id and false when not matched then insert (id, name) values (v.id, v.name)

# NULLs in the join key never match
create table nk (k int, v text)
insert into nk values (1, 'one'), (null, 'n1')
merge into nk t using (values (1, 'x1'), (null, 'n2')) as s(k, v) on t.k = s.k when matched then update set v = s.v when not matched then insert (k, v) values (s.k, s.v)
select * from nk order by k nulls last, v
merge into nk t using (values (1, 'x2'), (null, 'n3')) as s(k, v) on t.k is not distinct from s.k when matched then update set v = s.v
select * from nk order by k nulls last, v

# identity, serial and sequence defaults, in the order the rows are given

create sequence sq start 100

# privileges: each action needs its own, and SELECT for the ON clause and the source

grant select on src to lx_mrg
set role lx_mrg

reset role

set role lx_mrg
