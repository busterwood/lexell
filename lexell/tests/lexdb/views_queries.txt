create schema rep
create table emp (id int primary key, name text not null, dept int, salary numeric(10,2), active boolean default true)
create table dept (id int primary key, title varchar(30))
insert into dept values (1, 'Eng'), (2, 'Ops'), (3, 'Empty')
insert into emp values (1, 'Ann', 1, 100.50, true), (2, 'Bob', 1, 90, true), (3, 'Cy', 2, 70, false), (4, 'Di', null, 50, true)
create view active_emp as select id, name, dept, salary from emp where active
create view emp_dept (emp, dept_title) as select e.name, d.title from emp e join dept d on d.id = e.dept
create view dept_pay as select d.title, count(e.id) as n, sum(e.salary) as total from dept d left join emp e on e.dept = d.id group by d.title
create view top_paid as select name, salary from active_emp where salary > 60
create view rep.summary as with t as (select dept, max(salary) as top from emp group by dept) select dept, top from t where dept is not null
create view colours as values (1, 'red'), (2, 'green')
