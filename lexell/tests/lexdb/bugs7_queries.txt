# The seventh bug slice (design/lexdb-robust.md, "The seventh bug slice"), the PostgreSQL door's side.
# item 2: a SELECT * view keeps its columns as at CREATE VIEW (PostgreSQL expands * then): a column added to its table does not show
@once create table v7 (a int, b int)
@once insert into v7 values (1, 2), (3, NULL)
@once create view v7a as select * from v7
@once create view v7b as select v7.*, a + 10 as x from v7
@once create table w7 (k int, m text)
@once insert into w7 values (1, 'x'), (1, 'y')
@once create view v7c as select v7.*, w7.* from v7 join w7 on w7.k = v7.a
@once create view v7d (p, q) as select * from v7
@once create view v7e as select * from v7a where b is not null
@once alter table v7 add column c int default 7
select * from v7a order by a
select * from v7b order by a
select * from v7c order by a, m
select * from v7d order by p
select * from v7e
select a, b from v7a where b = 2
select count(*) as n from v7c v join v7 t on t.a = v.a
select v.a, t.c from v7a v join v7 t on t.a = v.a order by 1
select column_name from information_schema.columns where table_name = 'v7c' order by ordinal_position
select * from v7a where a > 100
# item 5: SET TRANSACTION READ ONLY makes the block read-only (25006 for a write, at any time); READ WRITE before the first query only (25001
# after it); ISOLATION LEVEL after the first query: the same level is taken, another 25001
@once create table tt7 (a int)
@once insert into tt7 values (1)
@tx set transaction read only; insert into tt7 values (5)
@tx set transaction read only; show transaction_read_only
@tx set transaction read only; select count(*) as n from tt7
@tx select 1; set transaction read only; insert into tt7 values (6)
@tx insert into tt7 values (7); set transaction read only; update tt7 set a = 0
@tx insert into tt7 values (7); set transaction read only; select count(*) as n from tt7
@tx select 1; set transaction read write; show transaction_read_only
@tx set transaction read only; select 1; set transaction read write
@tx set transaction read only; set transaction read write; insert into tt7 values (8)
@tx set transaction read only; set transaction read write; insert into tt7 values (8); select count(*) as n from tt7
@tx set transaction isolation level serializable, read only; show transaction_isolation
@tx set transaction read only isolation level serializable; delete from tt7
@tx set transaction isolation level serializable; select 1; set transaction isolation level serializable; show transaction_isolation
@tx set transaction isolation level serializable; select 1; set transaction isolation level read committed
@tx set transaction isolation level read committed; select 1; set transaction isolation level read committed; show transaction_isolation
@tx show transaction_read_only
show transaction_read_only
select count(*) as n from tt7
# item 4 (the T-SQL door's INFORMATION_SCHEMA): information_schema.domains (information_schema's own) and column_privileges
select domain_schema, domain_name, data_type, character_maximum_length, numeric_precision, datetime_precision, domain_default, udt_name from information_schema.domains order by domain_name
@once create role r7b7
@once grant select, update on tt7 to r7b7
@once grant insert on tt7 to public
select grantee, table_name, column_name, privilege_type, is_grantable from information_schema.column_privileges where table_name = 'tt7' and grantee <> current_user order by grantee, privilege_type
select count(*) as n from information_schema.column_privileges where table_name = 'tt7' and grantee = current_user
@once revoke all on tt7 from r7b7
@once drop role r7b7
