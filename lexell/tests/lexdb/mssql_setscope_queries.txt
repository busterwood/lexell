# A SET in a batch run inside another (sp_executesql, EXEC of a string) or in a procedure lasts to its end, as SQL Server's
# (gen_mssql.py setscope): mssql-jdbc's SQLServerBulkCopy and useBulkCopyForBatchInsert send "sp_executesql N'SET FMTONLY ON
# SELECT * FROM [t] '" and no OFF. Each batch runs with lexdb_batch off and on (BOTH) unless @once.
@once create table fq (id int identity primary key, v int)
@once insert into fq (v) values (1), (2)
@once create table fx (id int primary key)
@once insert into fx values (1)
sp_executesql N'SET FMTONLY ON SELECT * FROM [fq] '
select count(*) as n from fq
exec sp_executesql N'SET FMTONLY ON SELECT * FROM [fq] '; select count(*) as n from fq
exec ('set fmtonly on select * from fq'); select count(*) as n from fq
exec sp_executesql N'set fmtonly on select * from fq set fmtonly off'; select count(*) as n from fq
exec sp_executesql N'select count(*) as inner_n from fq set fmtonly on'; select count(*) as n from fq
set fmtonly on; select * from fq; set fmtonly off; select count(*) as n from fq
exec sp_executesql N'set nocount on'; update fq set v = v where id = 1
exec sp_executesql N'set nocount on; update fq set v = v where id = 1'
exec sp_executesql N'set rowcount 1'; select v from fq order by v
exec sp_executesql N'set rowcount 1; select v from fq order by v'
exec sp_executesql N'set quoted_identifier off'; select "a" as x
exec sp_executesql N'set identity_insert fq on'; insert into fq (id, v) values (100, 5)
@once exec sp_executesql N'set implicit_transactions on'; insert into fx values (5); select @@trancount as tc
@once delete from fx where id = 5
@once exec sp_executesql N'set xact_abort on'; begin tran; insert into fx values (1)
@once select @@trancount as tc; if @@trancount > 0 rollback
@once create procedure p_fmt as begin set fmtonly on; select * from fq end
exec p_fmt; select count(*) as n from fq
@once create procedure p_nc as set nocount on
exec p_nc; update fq set v = v where id = 1
@once create procedure p_ii as set identity_insert fq on
exec p_ii; insert into fq (id, v) values (101, 5)
@once create procedure p_it as set implicit_transactions on
@once exec p_it; insert into fx values (6); select @@trancount as tc
@once delete from fx where id = 6
select count(*) as n from fq
