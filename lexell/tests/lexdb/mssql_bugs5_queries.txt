# The fifth bug slice (design/lexdb-robust.md, "The fifth bug slice"): ALTER TABLE ... DROP COLUMN of a table views read is done (a view that used
# the column fails when read: 207, 4502 for SELECT *), unless a view made WITH SCHEMABINDING uses it (5074); WITH SCHEMABINDING, VIEW_METADATA and
# ENCRYPTION are read (195 for another; 1054 for SELECT * in a schema-bound view), sys.sql_modules.is_schema_bound
@once CREATE TABLE d1 (a int, b int, c int); INSERT INTO d1 VALUES (1, 2, 3), (4, NULL, 6)
@once CREATE VIEW d1a AS SELECT a FROM d1
@once CREATE VIEW d1b AS SELECT a, b FROM d1
@once CREATE VIEW d1s AS SELECT * FROM d1
@once CREATE VIEW d1vb AS SELECT a FROM d1b
@once CREATE VIEW d1cnt AS SELECT COUNT(*) AS n FROM d1
@once CREATE VIEW d1sb WITH SCHEMABINDING AS SELECT a, c FROM dbo.d1
@once CREATE VIEW d1sx WITH SCHEMABINDING, VIEW_METADATA AS SELECT a + c AS x FROM dbo.d1 WHERE c > 0
@once CREATE VIEW d1e WITH ENCRYPTION AS SELECT a FROM d1
CREATE VIEW d1bad WITH bogus AS SELECT a FROM d1
CREATE VIEW d1st WITH SCHEMABINDING AS SELECT * FROM dbo.d1
SELECT OBJECT_NAME(object_id) AS n, is_schema_bound FROM sys.sql_modules WHERE OBJECT_NAME(object_id) LIKE 'd1%' ORDER BY n
ALTER TABLE d1 DROP COLUMN c
SELECT * FROM d1sx ORDER BY x
@once ALTER TABLE d1 DROP COLUMN b
SELECT a FROM d1a ORDER BY a
SELECT n FROM d1cnt
