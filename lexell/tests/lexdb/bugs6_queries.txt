# The sixth bug slice (design/lexdb-robust.md, "The sixth bug slice"), the PostgreSQL door's side: DROP TABLE under views is refused (2BP01; the
# SQL Server and Oracle doors drop it, tds_bugs6 and tns_bugs6) and CASCADE drops the views (and the views over them); information_schema's
# view_table_usage and view_column_usage
@once create table d6 (a int, b int)
@once insert into d6 values (1, 2)
@once create view d6v as select a from d6
@once create view d6vv as select a from d6v
@once create table e6 (a int, c int)
@once create view e6v as select d6.a, e6.c from d6 join e6 on e6.a = d6.a
drop table d6
@tx drop table d6 cascade; select string_agg(table_name, ',' order by table_name) as v from information_schema.views where table_schema = 'public' and table_name like '%6v%'
@tx drop table e6 cascade; select string_agg(table_name, ',' order by table_name) as v from information_schema.views where table_schema = 'public' and table_name like '%6v%'
select view_schema, view_name, table_schema, table_name from information_schema.view_table_usage where view_name like '%6v%' order by view_name, table_name
select view_name, table_name, column_name from information_schema.view_column_usage where view_name like '%6v%' order by view_name, table_name, column_name
@once create view f6v as select * from d6
select view_name, table_name, column_name from information_schema.view_column_usage where view_name = 'f6v' order by column_name
