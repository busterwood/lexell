# against PostgreSQL 19 (gen_pg.py ... winnulls). Each statement runs with lexdb_batch off and on ("@once" lines once).
@once create table wn (id int primary key, g int, k text, x int, s text)
@once insert into wn values (1, 1, 'a', null, null), (2, 1, 'a', 10, 'p'), (3, 1, 'b', null, null), (4, 1, 'b', 20, 'q'), (5, 1, 'b', null, 'r'), (6, 2, 'a', null, null), (7, 2, 'a', null, null), (8, 2, 'c', 30, null), (9, 2, 'c', null, 's')
@once create table wb (id int primary key, g int, k text, x int)
@once insert into wb select i, i % 7, case when i % 5 = 0 then null else 'k' || (i % 3) end, case when i % 4 = 0 or i % 9 = 0 then null else i % 50 end from generate_series(1, 3000) i
@once create table we (id int primary key, x int)

select id, first_value(x) ignore nulls over (order by id) a, last_value(x) ignore nulls over (order by id) b, nth_value(x, 2) ignore nulls over (order by id rows between unbounded preceding and unbounded following) c, first_value(x) respect nulls over (order by id) d, last_value(x) ignore nulls over (order by id rows between 1 preceding and 1 following) e from wn order by id

select id, first_value(s) ignore nulls over (partition by k order by id desc) a, last_value(s) ignore nulls over (order by k, id range between unbounded preceding and unbounded following) b, lag(s, 1, 'none') ignore nulls over (order by id) c, nth_value(s, 3) ignore nulls over (order by id) d from wn order by id

select k, x, last_value(x) ignore nulls over (partition by k order by x nulls first) a, first_value(x) ignore nulls over (order by x desc nulls first, id) b from wn order by id
