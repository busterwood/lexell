# EXECUTE on functions and procedures, compared with PostgreSQL (tests/lexdb/gen_pg.py routinepriv). Run as a superuser.
drop role if exists lx_alice, lx_bob, lx_team
create role lx_alice login
create role lx_bob login
create role lx_team
create table t (id int primary key, v int)
insert into t values (1, 10), (2, 20)
create function f1(a int) returns int language sql as $$ select a + 1 $$
create function f2(a int) returns int language sql as $$ select v from t where id = a $$
create procedure p1(a int) language sql as $$ insert into t values (a, a * 10) $$
create schema s1
create function s1.f3() returns int language sql as $$ select 3 $$
create function f1(a int, b int) returns int language sql as $$ select a + b $$
set role lx_alice
select f1(1)
select f1(1, 2)
select f2(1)
call p1(5)
create function mine() returns int language sql as $$ select 1 $$
reset role
select proname, proacl is null from pg_proc where proname in ('f1', 'f2', 'p1') order by proname, pronargs
revoke execute on function f1(int) from public
set role lx_alice
select f1(1)
select f1(1, 2)
reset role
grant execute on function f1(int) to lx_alice
set role lx_alice
select f1(1)
select f1(1) + f1(2, 3)
select 1 where f1(2) = 3
reset role
set role lx_bob
select f1(1)
reset role
revoke execute on procedure p1(int) from public
set role lx_alice
call p1(5)
reset role

call p1(5)

select routine_name, privilege_type, is_grantable from information_schema.routine_privileges where routine_schema = 'public' and grantee = current_user order by routine_name
select grantee, routine_schema, routine_name, privilege_type from information_schema.role_routine_grants where routine_schema = 'public' and grantee <> current_user order by routine_name, grantee, privilege_type
select proname, proacl is null from pg_proc where proname in ('f1', 'f2', 'p1') order by proname, pronargs

select has_function_privilege('lx_alice', 'f1(int, int)', 'execute'), has_function_privilege('lx_alice', 'public.f1(integer, integer)', 'EXECUTE'), has_function_privilege('lx_alice', 's1.f3()', 'execute')
