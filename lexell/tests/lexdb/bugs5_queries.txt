# The fifth bug slice (design/lexdb-robust.md, "The fifth bug slice"), the PostgreSQL door's side: ALTER TABLE ... DROP COLUMN of a table views
# read is refused only when a view uses the column (2BP01: by name, through SELECT *, in WHERE, through a view over such a view), and CASCADE
# drops those views and no others
@once create table d1 (a int, b int, c int)
@once insert into d1 values (1, 2, 3), (4, null, 6)
@once create view d1a as select a from d1
@once create view d1b as select a, b from d1
@once create view d1s as select * from d1
@once create view d1w as select a from d1 where b > 0
@once create view d1vb as select a from d1b
@once create view d1cnt as select count(*) as n from d1
alter table d1 drop column b
alter table d1 drop column c
alter table d1 drop column a
@tx alter table d1 drop column b cascade; select string_agg(table_name, ',' order by table_name) as v from information_schema.views where table_schema = 'public' and table_name like 'd1%'
@tx drop view d1s; alter table d1 drop column c; select * from d1 order by a
@tx drop view d1s; alter table d1 drop column c; select * from d1w order by a
@tx alter table d1 drop column b cascade; select * from d1a order by a
@tx alter table d1 drop column b cascade; select n from d1cnt
@once alter table d1 add column d int
@once create view d1d as select a, d from d1
alter table d1 drop column d
@tx alter table d1 drop column d cascade; select string_agg(table_name, ',' order by table_name) as v from information_schema.views where table_schema = 'public' and table_name like 'd1%'
select string_agg(table_name, ',' order by table_name) as v from information_schema.views where table_schema = 'public' and table_name like 'd1%'
