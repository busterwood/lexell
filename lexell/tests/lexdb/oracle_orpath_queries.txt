# The OR of index paths (design/lexdb-optimizer.md 52: a table under an OR of indexed branches read by the branches' keys), as lexdb over TNS runs it. One entry a line: q or x, label, statement (tab separated); gen_oracle.py --dsn ... orpath, checked with --check. (No statistics: lexdb's TNS door has no ANALYZE TABLE ... COMPUTE STATISTICS and the test runs no automatic analyzer, so only the arms of unique keys are read by their keys; the others are answered by scans.)
x	table	create table o_t (id number(9) primary key, u number(9) not null, a number(9), b number(9), c varchar2(10), v number(9), w varchar2(20))
x	rows	insert into o_t select level, level * 3, case when mod(level, 97) = 0 then null else mod(level, 400) end, mod(level * 13, 1000), 'c' || mod(level, 10), mod(level * 31, 1009), 'w' || mod(level, 300) from dual connect by level <= 10000
x	unique	create unique index o_t_u on o_t (u)
x	index a	create index o_t_a on o_t (a)
x	index bc	create index o_t_bc on o_t (b, c)
x	index w	create index o_t_w on o_t (w)
x	table p	create table o_p (id number(9) primary key, g number(9), tag varchar2(10))
x	rows p	insert into o_p select level - 1, mod(level - 1, 1000), 't' || mod(level - 1, 7) from dual connect by level <= 1000
q	unique arms	select id, u, a from o_t where id in (3, 5, 7) or u = 36 or u = 30000 order by id
q	unique arms 2	select count(*) as n, sum(v) as s from o_t where id = 12 or u = 36 or id = -1
q	two indexes	select count(*) as n, sum(id) as s, min(id) as lo, max(id) as hi from o_t where a = 5 or b = 7
q	rows	select id, a, b from o_t where a = 5 or b = 7 order by id
q	in lists	select count(*) as n, sum(id) as s from o_t where a in (1, 2, 3) or b in (4, 5)
q	overlapping	select count(*) as n, sum(id) as s from o_t where a = 5 or (a = 5 and b < 500) or a between 4 and 6
q	primary key	select count(*) as n, sum(id) as s from o_t where id < 40 or a = 9 or id between 1000 and 1010
q	composite	select id, b, c from o_t where (b = 3 and c > 'c5') or (b = 4 and c = 'c1') or a = 77 order by id
q	null arm	select count(*) as n from o_t where a = null or b = 3
q	is null	select count(*) as n, sum(id) as s from o_t where a is null or b = 3
q	empty arm	select count(*) as n from o_t where a = -5 or b = 9
q	text	select id, w from o_t where w = 'w17' or w = 'w23' or a = 3 order by id fetch first 12 rows only
q	fetch first	select id from o_t where a = 5 or b = 7 order by id fetch first 5 rows only
q	group by	select a, count(*) as n from o_t where a in (5, 6) or b = 7 group by a order by a
q	join	select count(*) as n, sum(o_t.v) as s, min(o_p.tag) as m from o_t join o_p on o_p.id = o_t.b where o_t.a = 5 or o_t.b = 7
q	left join on	select count(*) as n, count(o_t.id) as m from o_p left join o_t on o_t.id = o_p.g and (o_t.a = 5 or o_t.b = 7) where o_p.id < 300
q	left join where	select count(*) as n, sum(o_t.id) as s from o_t left join o_p on o_p.id = o_t.b where o_t.a = 5 or o_t.b = 7
q	exists	select count(*) as n from o_p where exists (select 1 from o_t where o_t.a = o_p.g or o_t.b = o_p.g) and o_p.id < 50
q	two ors	select count(*) as n, sum(id) as s from o_t where (a = 5 or b = 7) and (a = 6 or b = 7 or b = 8)
q	division	select id from o_t where (a = 9 or b = 7 or id = 1009) and 10 / v > 0
x	update	update o_t set v = v + 1000 where a = 5 or b = 7
q	updated	select count(*) as n, sum(id) as s from o_t where v >= 1000
x	drop	drop table o_t purge
x	drop p	drop table o_p purge
