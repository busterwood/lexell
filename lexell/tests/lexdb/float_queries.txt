create table m (id int primary key, x float8, y real, z double precision)
insert into m values (1, 1.5, 0.1, 2), (2, -2.25, 1e7, null), (3, 'NaN', '-0', 1e300), (4, '-Infinity', 3.4e38, 0.5), (5, 0, 16777217, 1.5)
select * from m order by id
select id from m order by x, id
select id from m order by y desc, id
select id, x from m where x > 0 order by id
select id from m where x = 'nan' order by id
select sum(x), sum(y), sum(z), avg(z), pg_typeof(sum(y)), pg_typeof(avg(y)), pg_typeof(avg(z)), min(x), max(x), min(y), max(y) from m
select sum(z), avg(z), variance(z), stddev(z), var_pop(z) from m where id in (1, 4, 5)
select count(*), min(x), max(x) from m where id <> 3
select avg(y) from m where id in (1, 2)
create table fk (k float8 primary key, v text)
insert into fk values (1.5, 'a'), (-0.0, 'zero'), ('nan', 'n'), ('-inf', 'lo'), (1e300, 'big')
select k, v from fk order by k
insert into fk values (0, 'dup')
select v from fk where k = 'NaN'
select v from fk where k > 1 order by k
