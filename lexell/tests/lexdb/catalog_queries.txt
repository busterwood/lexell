# pg_catalog, information_schema and the SQL their clients use, compared with PostgreSQL (tests/lexdb/gen_pg.py catalog).
create schema shop
create table author (id serial primary key, name varchar(50) not null unique, born date, rating numeric(4,1) default 0, score int default -1 check (score >= -1))
create table book (id bigint generated always as identity primary key, title text not null default 'untitled', author_id int references author(id) on delete cascade, price numeric(10,2), tags text[], qty int, constraint qty_pos check (qty > 0))
create index book_title on book (title)
create unique index book_author_title on book (author_id, title) where price is not null
create table shop.orders (id int primary key, book_id bigint references book(id) on update restrict, qty int default 1, at timestamptz default now(), active boolean default true)
select table_schema, table_name, table_type from information_schema.tables where table_schema in ('public', 'shop') order by 1, 2
select table_name, column_name, ordinal_position, column_default, is_nullable, data_type, character_maximum_length, numeric_precision, numeric_precision_radix, numeric_scale, datetime_precision, udt_name, is_identity, identity_generation from information_schema.columns where table_schema in ('public', 'shop') order by table_name, ordinal_position
select constraint_name, table_schema, table_name, constraint_type from information_schema.table_constraints where table_schema in ('public', 'shop') and constraint_type <> 'CHECK' order by 1
select constraint_name, table_name, column_name, ordinal_position, position_in_unique_constraint from information_schema.key_column_usage where table_schema in ('public', 'shop') order by 1, 4
select constraint_name, unique_constraint_name, match_option, update_rule, delete_rule from information_schema.referential_constraints where constraint_schema in ('public', 'shop') order by 1

select sequence_schema, sequence_name, data_type, numeric_precision, start_value, minimum_value, maximum_value, increment, cycle_option from information_schema.sequences order by 2

select c.relname, a.attname, a.attnum, format_type(a.atttypid, a.atttypmod), a.attnotnull, a.atthasdef, a.attidentity, a.attndims from pg_attribute a join pg_class c on c.oid = a.attrelid join pg_namespace n on n.oid = c.relnamespace where n.nspname in ('public', 'shop') and a.attnum > 0 and c.relkind = 'r' order by 1, 3
select c.relname, a.attname, pg_get_expr(d.adbin, d.adrelid) from pg_attrdef d join pg_class c on c.oid = d.adrelid join pg_attribute a on a.attrelid = d.adrelid and a.attnum = d.adnum order by 1, 2
select conname, contype, pg_get_constraintdef(oid), conkey, confkey, confupdtype, confdeltype from pg_constraint where conrelid in ('author'::regclass, 'book'::regclass, 'shop.orders'::regclass) and contype in ('p', 'u', 'f') order by 1
select conname, pg_get_constraintdef(oid, true) from pg_constraint where conrelid = 'book'::regclass and contype = 'c' order by 1
select confrelid::regclass, conrelid::regclass from pg_constraint where contype = 'f' order by conname
select schemaname, tablename, indexname, indexdef from pg_indexes where schemaname in ('public', 'shop') order by 3

select a.attname from pg_index i, unnest(i.indkey) k(n) join pg_attribute a on a.attnum = k.n where i.indexrelid = 'book_author_title'::regclass and a.attrelid = 'book'::regclass order by 1
