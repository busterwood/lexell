# A SQL function called in a statement of another function's body or of a procedure (PostgreSQL has no DO and lexdb no PL/pgSQL: a SQL function
# and a procedure are the places where a statement runs inside another routine), against PostgreSQL 18 (gen_pg.py ... sqlfunc); each statement runs
# with lexdb_batch off and on ("@once" lines once, "@tx" ones in a transaction rolled back each time).
@once create table sf (a int primary key, c int)
@once insert into sf values (1, 10), (2, 20), (3, 30)
@once create function sf_f(x int) returns int language sql as $$ select x * 2 $$
@once create function sf_g(x int) returns int language sql as $$ select (select sf_f(a) from sf where a = x) $$
@once create function sf_h(x int) returns int language sql begin atomic select sf_f(x) + 1; end
@once create function sf_sum() returns bigint language sql as $$ select sum(sf_f(a)) from sf $$
@once create function sf_t(x int) returns text language sql as $$ select 'n' || sf_f(x) $$
@once create procedure sf_p1() language sql as $$ insert into sf select sf_f(a) + 100, 3 from sf where a = 1; update sf set c = sf_f(c) where a = 2 $$
@once create procedure sf_p2(x int) language sql as $$ delete from sf where a = sf_f(x) $$
select sf_f(a) from sf order by a
select sf_g(1), sf_h(2), sf_sum()
select a, sf_t(a) from sf where sf_f(a) > 2 order by a
@tx call sf_p1(); select a, c from sf order by a
@tx call sf_p2(1); select a, c from sf order by a
@tx update sf set c = sf_h(a) where sf_f(a) = 4; select a, c from sf order by a
@tx insert into sf select sf_h(a) + 1000, sf_f(a) from sf; select a, c from sf order by a
@tx delete from sf where a = sf_g(1); select a, c from sf order by a
select sf_g(7)
select sf_f(null)
select sf_f(a) from sf where false
