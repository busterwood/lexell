# array_agg, uuid and bytea, compared with PostgreSQL (tests/lexdb/gen_pg.py types).
create table pets (id int primary key, kind text, name text, born date)
insert into pets values (1, 'cat', 'Tom', '2020-01-01'), (2, 'dog', 'Rex', '2019-05-05'), (3, 'cat', null, '2021-03-03'), (4, 'dog', 'Ace', null)
select array_agg(name) from pets
select array_agg(name order by name) from pets
select array_agg(name order by name desc nulls last) from pets
select kind, array_agg(id order by id desc) from pets group by kind order by kind
select array_agg(distinct kind) from pets
select array_agg(born order by born) from pets
select array_agg(name) filter (where kind = 'cat') from pets
select array_agg(id) from pets where id > 10
select array_agg(name) from pets where id = 3

create table things (id uuid primary key, label text)
insert into things values ('00000000-0000-0000-0000-000000000002', 'two'), ('FFFFFFFF-0000-0000-0000-000000000000', 'ff'), ('{00000000-0000-0000-0000-000000000001}', 'one')
select * from things order by id
select label from things where id = '00000000000000000000000000000001'
select label from things where id > '00000000-0000-0000-0000-000000000001' order by id
insert into things values ('00000000-0000-0000-0000-000000000002', 'again')
select id::text, length(id::text) from things where label = 'ff'
select length(gen_random_uuid()::text), substr(gen_random_uuid()::text, 15, 1), gen_random_uuid() <> gen_random_uuid()
select substr(gen_random_uuid()::text, 20, 1) in ('8', '9', 'a', 'b')

create table blobs (k bytea primary key, n int)
insert into blobs values ('\x02', 2), ('\x01ff', 1), ('\x0100', 0), ('\x', -1)
select k, n from blobs order by k
select n from blobs where k = '\x01ff'
select n from blobs where k < '\x02' order by k desc
select encode(k, 'hex') from blobs order by k
