create function grp(gg text, out total int, out n bigint) returns setof record language sql as $$ select sum(v)::int, count(*) from t where g = gg $$

create function lit() returns setof text language sql as $$ values ('x'), ('y') $$
create function inc(a int default 1, b int default 100) returns table (s int) language sql as $$ select a + b $$
