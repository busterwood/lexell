# A scalar function called in a statement inside a batch with control flow, a procedure and EXEC (the T-SQL counterpart of the Oracle bug of
# KNOWN-ISSUES.md: a stored function in the SQL of a block), against SQL Server 2022 (gen_mssql.py ... udfblock). Each batch runs with SET lexdb_batch
# off and on.
@once CREATE TABLE ub (a int PRIMARY KEY, c int)
@once INSERT INTO ub VALUES (1, 10), (2, 20), (3, 30)
@once CREATE FUNCTION dbo.ubf(@x int) RETURNS int AS BEGIN RETURN @x * 2 END
@once CREATE FUNCTION dbo.ubg(@x int) RETURNS int AS BEGIN DECLARE @r int = 0; IF @x > 1 SET @r = @x * 3 ELSE SET @r = 1; RETURN @r END
@once CREATE PROCEDURE dbo.ubp @x int AS BEGIN DECLARE @v int; SELECT @v = dbo.ubf(a) FROM ub WHERE a = @x; INSERT INTO ub VALUES (@v + 100, 0); INSERT INTO ub VALUES (dbo.ubf(5), 3); UPDATE ub SET c = dbo.ubf(c) WHERE a = @x END
SELECT dbo.ubf(a) AS y FROM ub ORDER BY a
BEGIN TRANSACTION; DECLARE @v int; SELECT @v = dbo.ubf(a) FROM ub WHERE a = 1; INSERT INTO ub VALUES (@v + 100, 0); SELECT @v AS v; SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; IF 1 = 1 BEGIN DECLARE @v int; SELECT @v = dbo.ubg(a) FROM ub WHERE a = 2; INSERT INTO ub VALUES (@v + 100, 1) END; SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; EXEC dbo.ubp 1; SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; DECLARE @i int = 0; WHILE @i < 2 BEGIN INSERT INTO ub SELECT dbo.ubf(a) + 200 + @i, 4 FROM ub WHERE a = 1; SET @i = @i + 1 END; SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; EXEC('DECLARE @v int; SELECT @v = dbo.ubf(a) FROM ub WHERE a = 3; INSERT INTO ub VALUES (@v + 300, 5)'); SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; EXEC sp_executesql N'UPDATE ub SET c = dbo.ubg(a) WHERE a = @p', N'@p int', 3; SELECT a, c FROM ub ORDER BY a; ROLLBACK
BEGIN TRANSACTION; DELETE FROM ub WHERE a = dbo.ubf(1); SELECT a, c FROM ub ORDER BY a; ROLLBACK
