# IGNORE NULLS and RESPECT NULLS of LAG, LEAD, FIRST_VALUE and LAST_VALUE (SQL Server 2022; design/lexdb-new-features.md S15),
# checked against SQL Server 2022. One batch a line: gen_mssql.py winnulls. Each batch runs twice against lexdb (SET
# lexdb_batch = off, then on); "@once " lines (setup) run once. SQL Server's LAG and LEAD IGNORE NULLS take the row the
# offset names and, when its value is NULL, the nearest one not NULL beyond it (the default only past the partition).
@once create table wn (id int primary key, g int, k varchar(5), x int, s varchar(5)); insert into wn values (1, 1, 'a', null, null), (2, 1, 'a', 10, 'p'), (3, 1, 'b', null, null), (4, 1, 'b', 20, 'q'), (5, 1, 'b', null, 'r'), (6, 2, 'a', null, null), (7, 2, 'a', null, null), (8, 2, 'c', 30, null), (9, 2, 'c', null, 's')
@once create table wb (id int primary key, g int, k varchar(5), x int); insert into wb select value, value % 7, case when value % 5 = 0 then null else 'k' + cast(value % 3 as varchar(2)) end, case when value % 4 = 0 or value % 9 = 0 then null else value % 50 end from generate_series(1, 3000)
@once create table we (id int primary key, x int)
select id, lag(x) ignore nulls over (order by id) a, lead(x) ignore nulls over (order by id) b, lag(x, 2) ignore nulls over (order by id) c, lag(x, 0) ignore nulls over (order by id) d, lead(x, 1, -1) ignore nulls over (partition by g order by id) e from wn order by id

select id, first_value(x) ignore nulls over (order by id) a, last_value(x) ignore nulls over (order by id) b, first_value(x) respect nulls over (order by id) d, last_value(x) ignore nulls over (order by id rows between 1 preceding and 1 following) e from wn order by id
select id, first_value(x) ignore nulls over (order by id rows between 1 following and 3 following) a, last_value(x) ignore nulls over (partition by g order by id range between unbounded preceding and unbounded following) b, first_value(x) ignore nulls over (order by k, id rows between current row and unbounded following) c from wn order by id
select id, lag(s, 1, 'none') ignore nulls over (order by id) a, first_value(s) ignore nulls over (partition by k order by id desc) b, last_value(s) ignore nulls over (order by k, id range between unbounded preceding and unbounded following) c from wn order by id
select id, lag(x) ignore nulls over w1 a, first_value(x) ignore nulls over w1 b from wn window w1 as (partition by g order by id) order by id
select g, count(*) n, sum(a) sa, sum(b) sb, sum(c) sc, sum(d) sd, sum(f) sf from (select g, lag(x) ignore nulls over (partition by g order by id) a, lead(x, 3, 0) ignore nulls over (partition by g order by id) b, first_value(x) ignore nulls over (partition by g order by id rows between 5 preceding and 5 following) c, last_value(x) ignore nulls over (partition by k order by id) d, lag(x, 2, -1) respect nulls over (partition by g order by id) f from wb) t group by g order by g
select id, k, lag(x) ignore nulls over (partition by k order by id) a, last_value(x) ignore nulls over (partition by k order by id rows between 10 preceding and 2 preceding) b from wb where id in (5, 10, 11, 12, 1500, 2995, 2996, 3000) order by id
