drop role if exists lx_alice, lx_bob, lx_team
create role lx_alice login
create role lx_bob login
create role lx_team
create table acc (id int primary key, who text, amount numeric(10,2))
insert into acc values (1, 'a', 10), (2, 'b', 20)
create schema priv
create table priv.notes (id int primary key, body text)
insert into priv.notes values (1, 'x')
create view acc_small as select id, amount from acc where amount < 15
create sequence s1

select current_user, session_user = current_user
select * from acc

create table mine (id int primary key)

set role lx_alice
select id from acc order by id
update acc set amount = 0 where id = 1
select * from acc_small
reset role

revoke select on acc from lx_alice
set role lx_alice
select * from acc_small
select * from acc
reset role
set role lx_bob

select tablename, tableowner from pg_tables where tablename = 'mine'
select viewname, viewowner from pg_views where viewname = 'mine_v'
select has_table_privilege('lx_alice', 'acc', 'SELECT'), has_table_privilege('lx_bob', 'acc', 'INSERT'), has_table_privilege('lx_bob', 'acc', 'DELETE'), has_table_privilege('lx_alice', 'mine', 'DELETE'), has_table_privilege('lx_bob', 'mine', 'select, insert'), has_table_privilege('lx_alice', 'acc_small', 'select')
select has_schema_privilege('lx_bob', 'priv', 'USAGE'), has_schema_privilege('lx_alice', 'priv', 'USAGE'), has_schema_privilege('lx_alice', 'public', 'CREATE'), has_schema_privilege('lx_bob', 'public', 'CREATE, USAGE')
select grantee, privilege_type from information_schema.table_privileges where table_name = 'acc' and grantee like 'lx_%' order by 1, 2
select r.rolname, m.rolname from pg_auth_members a join pg_roles r on r.oid = a.roleid join pg_roles m on m.oid = a.member where r.rolname like 'lx_%' order by 1
select c.relname, pg_get_userbyid(c.relowner) from pg_class c where c.relname in ('mine', 'mine_v') order by 1
