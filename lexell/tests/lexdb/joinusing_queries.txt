# JOIN ... USING and NATURAL JOIN, compared with PostgreSQL (tests/lexdb/gen_pg.py joinusing).
create table emp (id int primary key, name text, dept int)
create table dept (dept int primary key, dname text)
create table pay (id int primary key, amt int, name text)
create table extra (code int primary key, note text, dept bigint)
insert into emp values (1, 'ann', 10), (2, 'bob', 20), (3, 'cy', null), (4, 'di', 30)
insert into dept values (10, 'eng'), (20, 'ops'), (40, 'hr')
insert into pay values (1, 100, 'ann'), (2, 200, 'robert'), (5, 500, 'eve')
insert into extra values (1, 'x', 10), (2, 'y', 40)
select * from emp join dept using (dept) order by id
select dept, name, dname from emp join dept using (dept) order by dept
select emp.dept, dept.dept, dept from emp join dept using (dept) order by 1
select * from emp left join dept using (dept) order by id
select * from emp right join dept using (dept) order by dept
select * from emp full join dept using (dept) order by dept, id
select dept, id from emp full join dept using (dept) order by dept nulls first, id
select dept, count(*) from emp full join dept using (dept) group by dept order by dept
select * from emp join pay using (id, name)
select * from emp join pay using (id) order by id
select id, emp.name, pay.name from emp join pay using (id) order by id
select name from emp join pay using (id) order by 1
select * from emp natural join pay
select * from emp natural left join pay order by id
select * from emp natural full join pay order by id, name
select * from emp natural join dept order by id
select count(*) from dept natural join pay
select * from emp join dept using (dept) join extra using (dept) order by id
select * from emp join dept using (dept) left join extra using (dept) order by id
select dept from emp join dept using (dept) where dept > 10 order by dept
select e.name, d.dname from emp e join dept d using (dept) where dept = 20
select * from emp e join dept d using (dept) order by e.id
select d.* from emp e join dept d using (dept) order by 1
select e.* from emp e join dept d using (dept) order by 1
select x.dept, x.dname from (select * from emp join dept using (dept)) x order by 1
create view ed as select * from emp join dept using (dept)
select * from ed order by id
select * from emp join dept using (nope)
select * from emp join pay using (dept)
select * from emp join dept using (dept, dept)
select * from emp cross join pay join dept using (dept) order by 1, 4
select * from emp, pay join emp e2 using (id) where false
select * from pay p1 cross join pay p2 join emp using (id)
select * from emp join dept on true join pay using (name) order by dname
select dept, (select count(*) from extra where extra.dept = dept) from emp join dept using (dept) order by 1
select * from emp natural join extra
update emp set name = upper(name) from dept join extra using (dept) where emp.dept = dept.dept
select id, name from emp order by id
select * from emp join dept using (dept) order by dept desc limit 1
select dept, sum(id) from emp join dept using (dept) group by 1 having dept < 30 order by 1
