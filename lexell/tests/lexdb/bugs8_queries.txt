# The eighth bug slice (design/lexdb-robust.md, "The eighth bug slice"), the PostgreSQL door's side.
# item 4: SET SESSION CHARACTERISTICS AS TRANSACTION READ ONLY | READ WRITE sets default_transaction_read_only: the session's later transactions
# (a block, a statement alone) are read-only (25006 for a write, a temporary table's writes apart); SET TRANSACTION READ WRITE overrides it
# before the block's first query; an isolation level alone leaves it; SET default_transaction_read_only and RESET do the same
@once create table ro8 (a int primary key, b text)
@once insert into ro8 values (1, 'x')
@once create temp table ro8t (a int)
show default_transaction_read_only
@once set session characteristics as transaction read only
show default_transaction_read_only
show transaction_read_only
select count(*) as n from ro8
insert into ro8 values (2, 'y')
update ro8 set b = 'z' where a = 1
delete from ro8 where a = 1
create table ro8b (a int)
truncate ro8
select * from ro8 for update
@once insert into ro8t values (1)
select count(*) as n from ro8t
@tx show transaction_read_only
@tx insert into ro8 values (3, 'w')
@tx set transaction read write; insert into ro8 values (3, 'w')
@tx set transaction read write; show transaction_read_only
@tx set transaction read only; show transaction_read_only
@tx select 1; set transaction read write
@tx insert into ro8t values (2)
@once set session characteristics as transaction isolation level serializable
show default_transaction_read_only
show default_transaction_isolation
@once set session characteristics as transaction read write
show default_transaction_read_only
@tx insert into ro8 values (2, 'y')
@tx show transaction_read_only
@once set default_transaction_read_only = on
show default_transaction_read_only
select count(*) as n from ro8
insert into ro8 values (2, 'y')
@tx show transaction_read_only
@once reset default_transaction_read_only
show default_transaction_read_only
@tx insert into ro8 values (2, 'y')
@once set session characteristics as transaction read only, isolation level serializable
show default_transaction_read_only
show default_transaction_isolation
insert into ro8 values (2, 'y')
@once set session characteristics as transaction read write, isolation level read committed
show default_transaction_read_only
@tx insert into ro8 values (2, 'y')
# item 4: a schema made AUTHORIZATION x is owned by x in information_schema.schemata
@once create role own8
@once create schema sch8 authorization own8
select schema_name, schema_owner from information_schema.schemata where schema_name = 'sch8'
select nspname, pg_get_userbyid(nspowner) as o from pg_namespace where nspname = 'sch8'
@once create schema sch8b authorization own8
select schema_name, schema_owner from information_schema.schemata where schema_name like 'sch8%' order by 1
@once drop schema sch8b
@once drop schema sch8
@once drop role own8
