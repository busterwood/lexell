# CREATE OR REPLACE VIEW
create table tv (id int primary key, v int, w text)
insert into tv values (1, 10, 'a'), (2, 20, 'b')
create view vv as select id, v from tv
select * from vv order by id
create or replace view vv as select id, v, w from tv
select * from vv order by id
create or replace view vv as select id, v, w from tv where id > 1
select * from vv order by id
create or replace view vv as select id, w from tv
create or replace view vv as select w, id, v from tv
create or replace view vv as select id, v::bigint as v, w from tv
create or replace view vv as select id, v, w, v * 2 as dbl from tv where id < 3
select * from vv order by id
create or replace view vnew as select id from tv
select * from vnew
create or replace view vv (a, b, c, d) as select id, v, w, v * 2 from tv
create or replace view vv (id, v, w, dbl) as select id, v, w, v + 1 from tv
select * from vv order by id
create or replace view tv as select 1 as a
create or replace view vv2 as select id, w from vv
create or replace view vv as select id, v, w, v + 1 as dbl, 'x' as extra from tv
select * from vv2 order by id
select * from vv order by id
create view vv as select id from tv
create or replace view vnone as select * from nothing_here
create or replace view vbad as select nothing from tv
create or alter view vca as select id from tv
create or replace table tnew (a int)
create or replace temp view vtemp as select id from tv
select * from vtemp order by id
create or replace temp view vtemp as select id, v from tv
select * from vtemp order by id
begin
create or replace view vv as select id, v, w, v + 1 as dbl, 'y' as extra from tv
select extra from vv where id = 1
rollback
